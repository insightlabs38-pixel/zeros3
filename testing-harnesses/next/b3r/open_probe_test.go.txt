func b3rRSS() int64 {
	b, _ := os.ReadFile("/proc/self/status")
	for _, l := range strings.Split(string(b), "\n") {
		if strings.HasPrefix(l, "VmHWM:") {
			var kb int64
			fmt.Sscanf(strings.TrimSpace(l[6:]), "%d", &kb)
			return kb << 10
		}
	}
	return 0
}

func TestB3ROpenProbe(t *testing.T) {
	dir, mode := os.Getenv("ZEROS3_B3R_STORE"), os.Getenv("ZEROS3_B3R_MODE")
	if dir == "" {
		t.Skip("probe")
	}
	switch mode {
	case "build":
		n := 150000
		s, err := OpenStore(dir)
		if err != nil {
			t.Fatal(err)
		}
		s.CreateBucket("b")
		var wg sync.WaitGroup
		for w := range 16 {
			wg.Go(func() {
				rng := rand.New(rand.NewPCG(uint64(w), 1))
				words := []string{"alpha", "beta", "gamma", "delta", "config", "value", "server", "index", "json", "token"}
				for i := w; i < n; i += 16 {
					var sb strings.Builder
					for sb.Len() < 600+rng.IntN(2000) {
						sb.WriteString(words[rng.IntN(len(words))])
						fmt.Fprintf(&sb, " %d\n", rng.IntN(1000))
					}
					k := fmt.Sprintf("tenant%02d/app%03d/obj%07d.json", i%37, i%211, i)
					if _, err := s.PutObject("b", k, []byte(sb.String()), "application/json", nil); err != nil {
						t.Error(err)
						return
					}
					if i%7 == 0 { // overwrite: retained history
						s.PutObject("b", k, []byte(sb.String()+"v2"), "application/json", nil)
					}
					if i%13 == 0 {
						s.DeleteObject("b", k)
					}
				}
			})
		}
		wg.Wait()
		s.Close()
		if _, err := compactStore(dir, defaultCompactOptions()); err != nil {
			t.Fatal(err)
		}
	case "publish":
		s, err := OpenStore(dir)
		if err != nil {
			t.Fatal(err)
		}
		s.nsKick()
		s.ns.wg.Wait()
		s.Close()
	case "open":
		runtime.GC()
		var a runtime.MemStats
		runtime.ReadMemStats(&a)
		t0 := time.Now()
		s, err := OpenStore(dir)
		if err != nil {
			t.Fatal(err)
		}
		el := time.Since(t0)
		var b runtime.MemStats
		runtime.ReadMemStats(&b)
		runtime.GC()
		var c runtime.MemStats
		runtime.ReadMemStats(&c)
		ck := s.ns.gen != nil && s.ns.gen.meta.Published
		rows := s.ns.objMut.n + s.ns.histMut.n
		// One small listing page and 1,000 lookups on the opened store.
		t1 := time.Now()
		page, _ := s.ListObjectsV2("b", "tenant07/", "/", "", 100)
		lt := time.Since(t1)
		t2 := time.Now()
		for i := range 1000 {
			s.lookupObject("b", fmt.Sprintf("tenant%02d/app%03d/obj%07d.json", (i*149)%37, (i*149)%211, i*149))
		}
		lk := time.Since(t2) / 1000
		out, _ := json.Marshal(map[string]any{"open_ms": el.Milliseconds(), "alloc_mb": float64(b.TotalAlloc-a.TotalAlloc) / 1e6,
			"live_heap_mb": float64(c.HeapAlloc) / 1e6, "peak_rss_mb": float64(b3rRSS()) / 1e6, "checkpoint": ck, "overlay_rows": rows,
			"list_us": lt.Microseconds(), "units": len(page.contents) + len(page.commonPrefixes), "lookup_us": float64(lk.Nanoseconds()) / 1e3})
		fmt.Println("RESULT", string(out))
		s.Close()
	}
}
