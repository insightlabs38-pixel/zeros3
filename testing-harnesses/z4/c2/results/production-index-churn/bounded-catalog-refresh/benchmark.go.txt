package main

import (
	"io"
	"os"
	"path/filepath"
	"testing"
)

func BenchmarkSimilarityCatalogSmallRefresh(b *testing.B) {
	s, err := OpenStoreShared(os.Getenv("ZEROS3_C2_INDEX_STORE"))
	if err != nil {
		b.Fatal(err)
	}
	defer s.Close()
	report, err := s.readAuthSimilarityCurrent()
	if err != nil {
		b.Fatal(err)
	}
	// Measure catalog copying separately from stage run hardlinks.
	report.Runs = nil
	catalog, err := os.Open(filepath.Join(report.Directory, "catalog.zsc"))
	if err != nil {
		b.Fatal(err)
	}
	var row [44]byte
	_, err = catalog.ReadAt(row[:], 12)
	catalog.Close()
	if err != nil {
		b.Fatal(err)
	}
	var first [32]byte
	copy(first[:], row[4:36])
	wanted := map[[32]byte]struct{}{first: {}}
	stage := b.TempDir()
	var remembered int
	b.ReportAllocs()
	for b.Loop() {
		lengths := make(map[[32]byte]int64)
		var retired []uint32
		if err := reuseAuthSimilarityCatalog(report, stage, io.Discard, lengths, wanted, false, &retired); err != nil {
			b.Fatal(err)
		}
		if _, found := lengths[first]; !found {
			b.Fatal("known touched digest was lost")
		}
		remembered = len(lengths)
	}
	b.ReportMetric(float64(remembered), "identities/op")
}
